cmake_minimum_required(VERSION 3.10)

# ==========================================
# 0. 交叉编译基础配置 (必须放在 project 之前)
# ==========================================
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_SYSTEM_PROCESSOR arm)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)

# 强制指定编译器 (建议在这里指定，防止 CMake 报错找不到编译器)
set(CMAKE_C_COMPILER arm-none-eabi-gcc)
set(CMAKE_ASM_COMPILER arm-none-eabi-gcc) # 汇编也需要编译器
set(CMAKE_OBJCOPY arm-none-eabi-objcopy)

# ==========================================
# 1. 项目定义
# ==========================================
project(stm32_cli_test C ASM)

# ==========================================
# [新增] 2. 对应你的 GCC 参数
# ==========================================

# 对应 -DSTM32F103xB
add_compile_definitions(STM32F103xB)

# 对应 -I./Drivers/...
# 使用 ${CMAKE_CURRENT_SOURCE_DIR} 确保路径绝对正确
include_directories(
    "${CMAKE_CURRENT_SOURCE_DIR}/Drivers/CMSIS/Core/Include"
    "${CMAKE_CURRENT_SOURCE_DIR}/Drivers/CMSIS/Device/ST/STM32F1xx/Include"
)

# ==========================================
# 3. 编译参数配置
# ==========================================
# 统一变量名为 MCU_FLAGS，方便下面调用
set(MCU_FLAGS "-mcpu=cortex-m3 -mthumb")

set(CMAKE_C_FLAGS "${MCU_FLAGS} -Wall -g -std=c99")
set(CMAKE_ASM_FLAGS "${MCU_FLAGS}") # 汇编文件(.s)也需要知道 CPU 架构

# ==========================================
# 4. 链接脚本配置
# ==========================================
set(LINKER_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/STM32F103XB_FLASH.ld")

# 注意：你原来这里写的是 ${MCU_FLAGS} 但定义的是 CPU_FLAGS，现在统一了
set(CMAKE_EXE_LINKER_FLAGS "${MCU_FLAGS} -T ${LINKER_SCRIPT} --specs=nosys.specs -Wl,-Map=${PROJECT_BINARY_DIR}/${PROJECT_NAME}.map -Wl,--gc-sections")

# ==========================================
# 5. 添加源文件 & 生成目标
# ==========================================
add_executable(${PROJECT_NAME}.elf 
    main.c
    # digital_num.c
    tm1650.c
    Drivers/CMSIS/Device/ST/STM32F1xx/Source/Templates/system_stm32f1xx.c
    startup_stm32f103xb.s  # 注意：你的宏是 xB，启动文件最好也用 xB 或 x8，不要用 x6
)

# ==========================================
# 6. 生成 .bin 文件
# ==========================================
add_custom_command(TARGET ${PROJECT_NAME}.elf POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O binary ${PROJECT_NAME}.elf ${PROJECT_NAME}.bin
    COMMENT "Building ${PROJECT_NAME}.bin"
)
